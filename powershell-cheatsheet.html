Ferramentas CLI Modernas

Substituem comandos antigos por versões mais rápidas e bonitas

cat→bat
Syntax highlighting, números de linha, git
bat arquivo.ps1
ls→eza
Ícones, cores, git status, tree
eza -la --icons --git
grep→ripgrep (rg)
10x mais rápido, .gitignore aware
rg "SELECT" --type sql
find→fd
Sintaxe simples, rápido
fd "\.log$" /var
git diff→diff-so-fancy
Diff humanamente legível
git diff | diff-so-fancy
man→tldr
Resumo prático de comandos
tldr tar

Ferramentas de Infra & Database

Específicas pro dia a dia de suporte, PostgreSQL e rede

pgcli
Cliente PostgreSQL com autocomplete inteligente e syntax highlighting. Substitui o psql.
pgcli -h servidor -U usuario -d banco
usql
Cliente SQL universal — PostgreSQL, MySQL, SQLite, SQL Server num comando só.
usql pg://user:pass@host/db
nmap
Scanner de rede. Descobre hosts, portas abertas, serviços, OS detection.
nmap -sP 192.168.1.0/24
WinMTR
Traceroute visual contínuo. Mostra perda de pacote por hop em tempo real.
winmtr
doggo
DNS lookup moderno. Substitui dig/nslookup com output colorido.
doggo google.com A MX
ping→gping
Ping gráfico de múltiplos hosts ao mesmo tempo.
gping google.com 8.8.8.8 1.1.1.1

# ═══════════════════════════════════════════════════════ # POWERSHELL PROFILE — fred's setup — dracula edition # ═══════════════════════════════════════════════════════ $env:POSH_THEMES_PATH = "$env:LOCALAPPDATA\Programs\oh-my-posh\themes" # --- Oh My Posh --- oh-my-posh init pwsh --config "$env:POSH_THEMES_PATH\powerlevel10k_rainbow.omp.json" | Invoke-Expression # --- PSReadLine --- Import-Module PSReadLine Set-PSReadLineKeyHandler -Key Tab -Function MenuComplete Set-PSReadLineOption -PredictionSource History Set-PSReadLineOption -PredictionViewStyle ListView Set-PSReadLineKeyHandler -Key UpArrow -Function HistorySearchBackward Set-PSReadLineKeyHandler -Key DownArrow -Function HistorySearchForward Set-PSReadLineKeyHandler -Key Ctrl+d -Function DeleteCharOrExit Set-PSReadLineKeyHandler -Key Ctrl+l -Function ClearScreen # --- Dracula Syntax Colors --- Set-PSReadLineOption -Colors @{ Command='#50fa7b';Parameter='#ffb86c';Operator='#ff79c6';Variable='#bd93f9' String='#f1fa8c';Number='#bd93f9';Type='#8be9fd';Comment='#6272a4' Keyword='#ff79c6';Error='#ff5555';Member='#8be9fd' InlinePrediction='#6272a4';ListPrediction='#bd93f9';Selection='#44475a' } # --- PSFzf --- Import-Module PSFzf Set-PsFzfOption -PSReadlineChordProvider 'Ctrl+t' Set-PsFzfOption -PSReadlineChordReverseHistory 'Ctrl+r' # --- Modules --- Import-Module Terminal-Icons Import-Module posh-git if (Get-Module -ListAvailable ZLocation) { Import-Module ZLocation } # ═══════════════════════════════════════════════════════ # ALIASES # ═══════════════════════════════════════════════════════ # Navegação (eza se disponível, senão fallback) if (Get-Command eza -ErrorAction SilentlyContinue) { function ll { eza -la --icons --git @args } function la { eza -a --icons @args } function lt { eza --tree --icons --level=2 @args } } else { function ll { Get-ChildItem -Force @args } function la { Get-ChildItem -Force -Hidden @args } } function .. { Set-Location .. } function ... { Set-Location ../.. } function ~ { Set-Location $HOME } function mkcd { param($dir) New-Item -ItemType Directory -Path $dir -Force; Set-Location $dir } # bat como cat (se instalado) if (Get-Command bat -ErrorAction SilentlyContinue) { function cat { bat @args } } # duf como df (se instalado) if (Get-Command duf -ErrorAction SilentlyContinue) { function df { duf @args } } else { function df { Get-PSDrive -PSProvider FileSystem } } # Git function gst { git status } function ga { git add @args } function gaa { git add --all } function gc { git commit -m @args } function gp { git push @args } function gl { git pull @args } function glog { git log --oneline --graph --decorate -20 } function gd { git diff @args } function gb { git branch @args } function gco { git checkout @args } # Arquivos function touch { param($file) if (Test-Path $file) { (Get-Item $file).LastWriteTime = Get-Date } else { New-Item $file -ItemType File } } function head { param($file, $n=10) Get-Content $file -TotalCount $n } function tail { param($file, $n=10) Get-Content $file -Tail $n } function which { Get-Command @args | Select-Object -ExpandProperty Source } function grep { Select-String @args } # Extract (equivalente do plugin extract do Zsh) function extract { param([string]$file) if (-not (Test-Path $file)) { Write-Host "Não encontrado: $file" -ForegroundColor Red; return } switch -Regex ($file) { '\.zip$' { Expand-Archive $file -DestinationPath (Split-Path $file) } '\.tar\.gz$|\.tgz$' { tar -xzf $file } '\.tar\.bz2$' { tar -xjf $file } '\.tar$' { tar -xf $file } '\.gz$' { gzip -d $file } '\.7z$' { 7z x $file } '\.rar$' { unrar x $file } default { Write-Host "Formato desconhecido" -ForegroundColor Yellow } } } # Sistema / Rede function myip { Invoke-RestMethod "https://api.ipify.org" } function ports { netstat -ano | Select-String "LISTENING" } # Clipboard (equivalente copypath / copyfile do Zsh) function copypath { (Get-Location).Path | Set-Clipboard; Write-Host "Copiado!" -ForegroundColor Green } function copyfile { param($file) Get-Content $file -Raw | Set-Clipboard; Write-Host "Conteúdo copiado!" -ForegroundColor Green } # Web search (equivalente web-search do Zsh) function google { param([string]$q) Start-Process "https://www.google.com/search?q=$([uri]::EscapeDataString($q))" } function ddg { param([string]$q) Start-Process "https://duckduckgo.com/?q=$([uri]::EscapeDataString($q))" } # Open (equivalente xdg-open / open do macOS) function open { param($path=".") Start-Process $path } # Boas-vindas Write-Host "PowerShell $($PSVersionTable.PSVersion) | $(oh-my-posh version)" -ForegroundColor DarkGray
